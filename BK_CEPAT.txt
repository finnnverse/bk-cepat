<script>
(function(){
  'use strict';
  var API = 'https://script.google.com/macros/s/AKfycbxF4OWQxWw-AO1Tq1Zq80eG75Oi5emHDYDjr_8QVoOHkvx365ew3XMhnBjlxf9yFbBdoQ/exec';
  var KDKMP = 'KDKMP-002';

  var $ = function(id){ return document.getElementById(id); };
  var barang = [];
  var dipilih = null;

  var sub = $('sub'), hasil = $('hasil'), panel = $('panel'),
      cari = $('cari'), kirim = $('kirim'), statusEl = $('status');

  function load(){
    fetch(API + '?action=getInitialData&kdkmpId=' + KDKMP)
      .then(function(r){ return r.json(); })
      .then(function(d){
        if(!d || !d.success){ throw new Error('Gagal ambil data'); }
        barang = (d.barang || []).slice();
        var namaUnit = (d.kdkmp && d.kdkmp.nama) ? d.kdkmp.nama : KDKMP;
        sub.textContent = KDKMP + ' (' + namaUnit + ') · ' + barang.length + ' barang siap';
        render(cari ? cari.value : '');
      })
      .catch(function(e){
        sub.textContent = 'Gagal ambil data — cek koneksi';
        hasil.innerHTML = '<div class="kosong">Gagal memuat barang. Cek internet, refresh.</div>';
      });
  }

  function fmtStok(x){ return (typeof x === 'number') ? x.toLocaleString('id-ID') : (x || '0'); }

  function render(kata){
    var q = kata.trim().toLowerCase();
    if(barang.length === 0){
      hasil.innerHTML = '<div class="kosong">Memuat daftar barang... tunggu sebentar</div>';
      return;
    }
    var f = q ? barang.filter(function(b){
      return (b.nama || '').toLowerCase().indexOf(q) > -1 ||
             (b.kode || '').toLowerCase().indexOf(q) > -1;
    }) : [];
    if(!q){
      hasil.innerHTML = '<div class="kosong">Ketik dulu buat cari barang…</div>';
      return;
    }
    if(!f.length){
      hasil.innerHTML = '<div class="kosong">Tidak ada barang cocok dengan "' + kata + '"</div>';
      return;
    }
    var tampil = f.slice(0, 60);
    var html = '';
    tampil.forEach(function(b){
      html += '<div class="item" data-kode="' + b.kode + '">' +
              '<div><div class="nm">' + escapeHtml(b.nama) + '</div>' +
              '<div class="kd">' + b.kode + (b.satuan ? ' · satuan: ' + b.satuan : '') + '</div></div>' +
              '<div class="st">stok ' + fmtStok(b.stokSystem) + '</div></div>';
    });
    if(f.length > tampil.length){
      html += '<div class="kosong">…dan ' + (f.length - tampil.length) + ' lainnya (perhalus kata cari)</div>';
    }
    hasil.innerHTML = html;
    hasil.querySelectorAll('.item').forEach(function(el){
      el.addEventListener('click', function(){
        pilih(barang.find(function(b){ return b.kode === el.getAttribute('data-kode'); }));
      });
    });
  }

  function escapeHtml(s){
    return String(s == null ? '' : s).replace(/&/g,'&amp;').replace(/</g,'&lt;').replace(/>/g,'&gt;').replace(/"/g,'&quot;');
  }

  function pilih(b){
    if(!b) return;
    dipilih = b;
    $('pilihNama').textContent = b.nama;
    $('pilihKode').textContent = b.kode;
    $('pilihStok').textContent = 'Stok sistem: ' + fmtStok(b.stokSystem) + ' ' + (b.satuan || '');
    panel.style.display = 'block';
    panel.scrollIntoView({ behavior: 'smooth', block: 'start' });
    $('jumlah').focus();
  }

  cari.addEventListener('input', function(){ render(cari.value); });
  cari.addEventListener('focus', function(){ render(cari.value); });

  kirim.addEventListener('click', function(){
    if(!dipilih){
      status('Pilih dulu barangnya dari hasil cari.', 'gagal'); return;
    }
    var jml = parseInt($('jumlah').value, 10);
    if(!jml || jml <= 0){
      status('Isi jumlah keluar (minimal 1).', 'gagal'); return;
    }
    if(typeof dipilih.stokSystem === 'number' && jml > dipilih.stokSystem){
      status('Stok gak cukup! Stok sistem: ' + dipilih.stokSystem + ' ' + (dipilih.satuan || ''), 'gagal'); return;
    }
    var payload = {
      action: 'submitBarangKeluar',
      kdkmpId: KDKMP,
      kodeBarang: dipilih.kode,
      jumlah: jml,
      keterangan: ($('ket').value || '').trim(),
      petugas: ($('petugas').value || '').trim() || 'Petugas'
    };
    kirim.disabled = true;
    kirim.textContent = 'Mengirim…';
    status('', 'hapus');
    fetch(API, {
      method: 'POST',
      headers: { 'Content-Type': 'text/plain;charset=utf-8' },
      body: JSON.stringify(payload)
    })
    .then(function(r){ return r.json(); })
    .then(function(res){
      kirim.disabled = false;
      kirim.textContent = 'Kirim ➜ Masuk ke Sistem';
      if(res && res.success){
        status('✅ Berhasil! ' + dipilih.nama + ' -' + jml + ' ' + (dipilih.satuan || '') + ' masuk ke riwayat.\nPesan: ' + (res.message || 'ok'), 'ok');
        $('jumlah').value = ''; $('ket').value = '';
        var b = dipilih;
        if(typeof b.stokSystem === 'number') b.stokSystem = Math.max(0, b.stokSystem - jml);
        $('pilihStok').textContent = 'Stok sistem: ' + fmtStok(b.stokSystem) + ' ' + (b.satuan || '');
      } else {
        status('❌ Gagal: ' + (res && res.message ? res.message : 'dari sistem'), 'gagal');
      }
    })
    .catch(function(e){
      kirim.disabled = false;
      kirim.textContent = 'Kirim ➜ Masuk ke Sistem';
      status('❌ Gagal terhubung ke sistem. Cek internet, coba lagi.', 'gagal');
    });
  });

  function status(teks, tipe){
    statusEl.textContent = teks;
    statusEl.className = tipe;
  }

  // ===== LAPORAN BARANG KELUAR =====
  var riwayat = [];
  var rptPeriode = 'semua';
  var RPT = {
    hari: 'Hari Ini', kemarin: 'Kemarin', '7hari': '7 Hari', semua: 'Keseluruhan'
  };

  function fmtTgl(t){
    var d = new Date(t);
    if(isNaN(d)) return String(t || '');
    // paksa WIB (UTC+7) biar konsisten laporannya
    var w = new Date(d.getTime() + 7*3600*1000);
    var bln = ['Jan','Feb','Mar','Apr','Mei','Jun','Jul','Agu','Sep','Okt','Nov','Des'];
    return w.getUTCDate() + ' ' + bln[w.getUTCMonth()] + ' ' + w.getUTCHours() + ':' + ('0'+w.getUTCMinutes()).slice(-2);
  }
  function tglIso(x){
    var d = new Date(x);
    if(isNaN(d)) return '';
    var w = new Date(d.getTime() + 7*3600*1000);
    return w.toISOString().slice(0,10);
  }
  function tglHariIni(){
    var w = new Date(Date.now() + 7*3600*1000);
    return w.toISOString().slice(0,10);
  }
  function dalamRpt(t){
    if(rptPeriode === 'semua') return true;
    var iso = tglIso(t);
    if(!iso) return false;
    if(rptPeriode === 'hari') return iso === tglHariIni();
    if(rptPeriode === 'kemarin'){
      var d = new Date(); d.setDate(d.getDate()-1);
      var km = d.getFullYear() + '-' + ('0'+(d.getMonth()+1)).slice(-2) + '-' + ('0'+d.getDate()).slice(-2);
      return iso === km;
    }
    if(rptPeriode === '7hari'){
      var a = new Date(tglHariIni() + 'T00:00:00'), b = new Date(iso + 'T00:00:00');
      return Math.round((a - b) / 86400000) <= 6;
    }
    return true;
  }
  function tagKet(k){
    k = String(k || '').toLowerCase();
    if(k.indexOf('jual') > -1 || k.indexOf('penjualan') > -1) return '<span class="tag jual">Penjualan</span>';
    if(k.indexOf('rusak') > -1) return '<span class="tag rusak">Rusak</span>';
    if(k.indexOf('exp') > -1) return '<span class="tag expired">Expired</span>';
    return '<span class="tag lain">' + escapeHtml(k || '-') + '</span>';
  }

  function renderLaporan(){
    var f = riwayat.filter(function(h){ return dalamRpt(h.tanggal); });
    var totPcs = f.reduce(function(a,h){ return a + (Number(h.jumlah)||0); }, 0);
    var petugas = {};
    f.forEach(function(h){ petugas[String(h.petugas||'?')] = 1; });

    $('rptPcs').textContent = totPcs.toLocaleString('id-ID') + ' pcs';
    $('rptTrx').textContent = f.length;
    $('rptPetugas').textContent = Object.keys(petugas).length;
    $('rptInfo').textContent = '· ' + f.length + (f.length === 1 ? ' catatan' : ' catatan');

    // rekap per barang (urut qty terbesar)
    var by = {};
    f.forEach(function(h){
      var k = String(h.kode || '');
      by[k] = by[k] || { nama: String(h.nama || k), pcs: 0, awal: null, akhir: null };
      by[k].pcs += Number(h.jumlah)||0;
      by[k].awal = (by[k].awal == null) ? h.stokAwal : by[k].awal;
      by[k].akhir = h.stokAkhir != null ? h.stokAkhir : by[k].akhir;
    });
    var arr = Object.keys(by).map(function(k){ return by[k]; })
      .sort(function(a,b){ return b.pcs - a.pcs; });

    var hTabel = '<table><thead><tr><th>Barang</th><th style="text-align:right">Qty</th>' +
      '<th style="text-align:right">Stok</th></tr></thead><tbody>';
    if(!arr.length){
      hTabel += '<tr><td colspan="3" style="text-align:center;color:#aaa;padding:18px">Belum ada barang keluar periode ini</td></tr>';
    } else {
      arr.forEach(function(a){
        var stok = (a.awal != null && a.akhir != null) ? a.awal + ' → ' + a.akhir : (a.akhir != null ? String(a.akhir) : '-');
        hTabel += '<tr><td class="ru">' + escapeHtml(a.nama) + '</td>' +
          '<td style="text-align:right" class="ru">' + a.pcs + '</td>' +
          '<td style="text-align:right;color:#888">' + stok + '</td></tr>';
      });
    }
    hTabel += '</tbody></table>';
    $('rptRekap').innerHTML = '<div style="font-size:12.5px;font-weight:800;margin-top:10px">Rekap per Barang</div>' + hTabel;

    // riwayat detail (terbaru di atas)
    var list = f.slice().sort(function(a,b){ return String(b.tanggal).localeCompare(String(a.tanggal)); });
    var hList = '<table><thead><tr><th>Waktu</th><th>Barang</th><th style="text-align:right">Qty</th>' +
      '<th>Petugas</th><th>Ket</th></tr></thead><tbody>';
    if(!list.length){
      hList += '<tr><td colspan="5" style="text-align:center;color:#aaa;padding:18px">Belum ada riwayat periode ini</td></tr>';
    } else {
      list.slice(0, 60).forEach(function(h){
        hList += '<tr><td style="white-space:nowrap;font-size:11.5px">' + fmtTgl(h.tanggal) + '</td>' +
          '<td class="ru">' + escapeHtml(String(h.nama||'').slice(0,40)) + '</td>' +
          '<td style="text-align:right" class="ru">' + (Number(h.jumlah)||0) + '</td>' +
          '<td style="font-size:12px">' + escapeHtml(String(h.petugas||'-').split(' ')[0]) + '</td>' +
          '<td>' + tagKet(h.keterangan) + '</td></tr>';
      });
      if(list.length > 60) hList += '<tr><td colspan="5" style="text-align:center;color:#aaa">+ ' + (list.length-60) + ' lagi (perhalus periode)</td></tr>';
    }
    hList += '</tbody></table>';
    $('rptList').innerHTML = '<div style="font-size:12.5px;font-weight:800;margin-top:12px">Riwayat Kirim</div>' + hList;
  }

  function buatTombolRpt(){
    var h = '';
    ['hari','kemarin','7hari','semua'].forEach(function(k){
      h += '<button data-r="' + k + '" class="' + (rptPeriode === k ? 'aktif' : '') + '">' + RPT[k] + '</button>';
    });
    $('rptPeriode').innerHTML = h;
    $('rptPeriode').querySelectorAll('button').forEach(function(b){
      b.addEventListener('click', function(){
        rptPeriode = b.getAttribute('data-r');
        $('rptPeriode').querySelectorAll('button').forEach(function(x){ x.classList.remove('aktif'); });
        b.classList.add('aktif');
        renderLaporan();
      });
    });
  }

  function muatRiwayat(){
    fetch(API + '?action=getBarangKeluarHistory&kdkmpId=' + KDKMP)
      .then(function(r){ return r.json(); })
      .then(function(d){
        if(!d || !d.success || !Array.isArray(d.history)) throw new Error('gagal');
        riwayat = d.history.slice();
        $('kartuLaporan').style.display = 'block';
        renderLaporan();
      })
      .catch(function(){ $('kartuLaporan').style.display = 'none'; });
  }

  buatTombolRpt();
  load();
  setTimeout(muatRiwayat, 500);
})();
</script>
</body>
</html>`; }
